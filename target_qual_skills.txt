                      const newSkill =
                        value.replace(",", "").trim();

                      if (

                        !skills.includes(newSkill)
                      ) {

                        setSkills([
                          ...skills,
                          newSkill,
                        ]);

                      }

                      setSkillInput("");

                      setSkillInput(value);

                    }

                  }}
